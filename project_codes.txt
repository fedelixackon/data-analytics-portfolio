#model evaluation
evaluate_model <- function(model_name, actual, predicted, probability) {
  
  actual <- factor(
    actual,
    levels = c("NoPurchase", "Purchase")
  )
  
  predicted <- factor(
    predicted,
    levels = c("NoPurchase", "Purchase")
  )
  
  cm <- table(
    Predicted = predicted,
    Actual = actual
  )
  
  TN <- cm["NoPurchase", "NoPurchase"]
  FN <- cm["NoPurchase", "Purchase"]
  FP <- cm["Purchase", "NoPurchase"]
  TP <- cm["Purchase", "Purchase"]
  
  accuracy <- (TP + TN) / sum(cm)
  
  sensitivity <- ifelse(
    TP + FN == 0,
    NA,
    TP / (TP + FN)
  )
  
  specificity <- ifelse(
    TN + FP == 0,
    NA,
    TN / (TN + FP)
  )
  
  precision <- ifelse(
    TP + FP == 0,
    NA,
    TP / (TP + FP)
  )
  
  f1_score <- ifelse(
    is.na(precision) |
      is.na(sensitivity) |
      precision + sensitivity == 0,
    NA,
    2 * precision * sensitivity /
      (precision + sensitivity)
  )
  
  roc_object <- roc(
    response = actual,
    predictor = probability,
    levels = c("NoPurchase", "Purchase"),
    direction = "<",
    quiet = TRUE
  )
  
  auc_value <- as.numeric(auc(roc_object))
  
  results <- data.frame(
    Model = model_name,
    Accuracy = accuracy,
    Sensitivity = sensitivity,
    Specificity = specificity,
    Precision = precision,
    F1_Score = f1_score,
    AUC = auc_value
  )
  
  return(
    list(
      confusion_matrix = cm,
      results = results
    )
  )
}

#logistic regression
  
logistic_model <- glm(
  Revenue ~ .,
  data = train_data,
  family = binomial
)

# Predicted probabilities
logistic_probability <- predict(
  logistic_model,
  newdata = test_data,
  type = "response"
)

# Convert probabilities into predicted classes
logistic_prediction <- ifelse(
  logistic_probability >= 0.50,
  "Purchase",
  "NoPurchase"
)

logistic_prediction <- factor(
  logistic_prediction,
  levels = c("NoPurchase", "Purchase")
)

# Evaluate the model
logistic_evaluation <- evaluate_model(
  model_name = "Logistic Regression",
  actual = test_data$Revenue,
  predicted = logistic_prediction,
  probability = logistic_probability
)

logistic_evaluation$confusion_matrix
logistic_evaluation$results
